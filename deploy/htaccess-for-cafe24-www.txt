# ############################################################################
# ##  주의 — 현재 lasagna.kr 호스팅에서는 이 파일이 동작하지 않는다.
# ##
# ##  서버가 openresty(nginx) 라서 .htaccess 를 아예 읽지 않는다.
# ##  .htaccess 는 Apache 전용 기능이다.
# ##
# ##  현재 상황의 해결책은 docs/CAFE24-HTTPS.md 참고.
# ##  이 파일은 나중에 Apache 호스팅으로 옮길 경우를 위해 남겨둔다.
# ############################################################################

# ============================================================================
#  라자냐 엠파이어 — HTTP→HTTPS + 주소를 www 로 통일 (카페24용)
#
#  [이 파일을 쓰는 경우]
#    · https://lasagna.kr (www 없이) 에서 인증서 경고가 뜬다
#      → 인증서가 www.lasagna.kr 만 보호한다는 뜻. 이 파일을 쓴다.
#    · 또는 주소를 www 하나로 통일하고 싶다 (검색엔진에 유리)
#
#  [중요] 서버에 올린 뒤 이름을 ".htaccess" 로 바꿔야 작동한다.
#  [되돌리기] 문제가 생기면 이 파일을 지우기만 하면 즉시 원래대로 돌아온다.
# ============================================================================

RewriteEngine On

# ---------------------------------------------------------------------------
# 규칙 1) www 가 없는 주소는 www 붙여서 https 로 보낸다.
#         lasagna.kr  →  https://www.lasagna.kr
#
# 이 규칙 덕분에 인증서가 www 만 보호해도 문제가 없다.
# 방문자가 http://lasagna.kr 로 들어와도, 인증서가 필요 없는 http 단계에서
# 곧바로 www 주소로 넘어가기 때문이다.
# ---------------------------------------------------------------------------
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^ https://www.%{HTTP_HOST}%{REQUEST_URI} [L,R=302]

# ---------------------------------------------------------------------------
# 규칙 2) 이미 www 인데 아직 http 라면 https 로 올린다.
#
# 조건을 두 가지로 확인하는 이유: 카페24 서버 구성에 따라 둘 중 하나만
# 인식되는 경우가 있는데, 하나만 쓰면 무한 리다이렉트에 빠질 수 있다.
# ("리디렉션이 너무 많습니다" 오류가 그것)
# ---------------------------------------------------------------------------
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=302]


# ============================================================================
#  테스트가 끝나면:  R=302  두 곳을 모두  R=301  로 바꾸고 다시 업로드.
#
#  302 = 임시(브라우저가 기억 안 함) · 301 = 영구(브라우저가 기억함)
#  302 로 먼저 확인하는 이유는, 설정이 틀렸을 때 301 은 파일을 지워도
#  한동안 계속 튕기기 때문이다.
# ============================================================================


# ============================================================================

# ============================================================================
