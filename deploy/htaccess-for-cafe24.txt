# ============================================================================
#  라자냐 엠파이어 — HTTP → HTTPS 리다이렉트 (카페24용)
#
#  [중요] 이 파일을 서버에 올린 뒤 이름을 ".htaccess" 로 바꿔야 작동한다.
#         앞에 점(.) 이 반드시 붙는다. 확장자는 없다.
#
#  [순서] SSL 인증서가 먼저 발급된 다음에 올려라.
#         인증서 없이 올리면 사이트가 안 열린다.
#
#  [되돌리기] 문제가 생기면 이 파일을 지우기만 하면 즉시 원래대로 돌아온다.
# ============================================================================

RewriteEngine On

# ---------------------------------------------------------------------------
# 아래 두 줄은 "지금 접속이 HTTPS 가 아닐 때만" 이라는 조건이다.
# 두 가지 방법으로 확인하는 이유: 카페24 서버 구성에 따라 둘 중 하나만
# 인식되는 경우가 있는데, 하나만 쓰면 무한 리다이렉트에 빠질 수 있다.
# (브라우저에 "리디렉션이 너무 많습니다" 오류가 뜨는 그 현상)
# ---------------------------------------------------------------------------
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https

# ---------------------------------------------------------------------------
# 조건에 맞으면 https:// 같은 주소로 보낸다.
#
# R=302  = 임시 이동. 브라우저가 기억하지 않는다.  ← 처음엔 이걸로 테스트
# R=301  = 영구 이동. 브라우저가 기억한다.         ← 잘 되면 이걸로 바꾼다
#
# 302 로 먼저 테스트하는 이유: 301 은 브라우저가 캐시에 저장해버려서,
# 설정이 틀렸을 때 파일을 지워도 한동안 계속 튕긴다. 302 는 그런 게 없다.
# ---------------------------------------------------------------------------
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=302]


# ============================================================================
#  테스트가 끝나고 잘 넘어가는 걸 확인했다면:
#    위 줄의  R=302  를  R=301  로 바꾸고 다시 업로드.
# ============================================================================


# ============================================================================
#  [나중에] HSTS — 지금은 켜지 마라.
#
#  아래 줄은 "이 사이트는 앞으로 1년간 무조건 HTTPS 로만 접속" 을 브라우저에
#  강제하는 설정이다. 보안은 올라가지만, 인증서에 문제가 생기면 사이트에
#  아예 접근할 수 없게 되고 되돌리는 데 시간이 오래 걸린다.
#
#  위 리다이렉트가 몇 주간 문제없이 돌아간 뒤에 켜는 걸 권한다.
#  켤 때는 아래 줄 맨 앞의 # 을 지우면 된다.
# ============================================================================
